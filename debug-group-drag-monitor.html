📊 监控状态

监控器: 未激活

📝 使用说明

  1. 在浏览器中打开你的应用（http://localhost:3000）
  2. 打开浏览器控制台（F12）
  3. 在控制台中粘贴并执行下面的代码
  4. 拖动组，观察日志输出
  5. 找到闪烁发生的时刻和原因

💻 监控代码（复制到浏览器控制台）

// 🔍 组拖动闪烁监控器
(function() {
    console.log('🔍 组拖动闪烁监控器已启动');
    
    // 查找所有组元素
    const groups = document.querySelectorAll('[data-group-id]');
    console.log(`📦 找到 ${groups.length} 个组`);
    
    if (groups.length === 0) {
        console.error('❌ 没有找到组元素！请确保页面已加载完成。');
        return;
    }
    
    // 监控每个组
    groups.forEach((group, index) => {
        const groupId = group.getAttribute('data-group-id');
        console.log(`\n📌 监控组 ${index + 1}: ${groupId}`);
        
        // 记录初始状态
        const initialState = {
            left: group.style.left,
            top: group.style.top,
            transform: group.style.transform,
            transition: group.style.transition,
            willChange: group.style.willChange
        };
        console.log('初始状态:', initialState);
        
        // 监控样式变化
        const observer = new MutationObserver((mutations) => {
            mutations.forEach(mutation => {
                if (mutation.attributeName === 'style') {
                    const timestamp = new Date().toISOString().split('T')[1];
                    const currentState = {
                        left: group.style.left,
                        top: group.style.top,
                        transform: group.style.transform,
                        transition: group.style.transition,
                        willChange: group.style.willChange
                    };
                    
                    // 检测关键变化
                    const changes = [];
                    if (currentState.transform !== initialState.transform) {
                        changes.push(`transform: "${initialState.transform}" → "${currentState.transform}"`);
                    }
                    if (currentState.transition !== initialState.transition) {
                        changes.push(`transition: "${initialState.transition}" → "${currentState.transition}"`);
                    }
                    if (currentState.left !== initialState.left) {
                        changes.push(`left: "${initialState.left}" → "${currentState.left}"`);
                    }
                    if (currentState.top !== initialState.top) {
                        changes.push(`top: "${initialState.top}" → "${currentState.top}"`);
                    }
                    
                    if (changes.length > 0) {
                        console.log(`\n⏰ [${timestamp}] 组 ${groupId} 样式变化:`);
                        changes.forEach(change => console.log(`  - ${change}`));
                        
                        // 🔥 检测闪烁的关键时刻
                        if (currentState.transition !== 'none' && currentState.transform === '') {
                            console.error(`🚨 闪烁警告！transition 已启用但 transform 被清除`);
                            console.error(`  - 这会导致从 transform 位置过渡到 left/top 位置`);
                            console.error(`  - 当前 transition: "${currentState.transition}"`);
                            console.error(`  - 当前 left: "${currentState.left}", top: "${currentState.top}"`);
                        }
                        
                        // 更新初始状态
                        Object.assign(initialState, currentState);
                    }
                }
            });
        });
        
        observer.observe(group, { 
            attributes: true, 
            attributeFilter: ['style'] 
        });
        
        console.log(`✅ 组 ${groupId} 监控已启动`);
    });
    
    // 监控子节点
    const nodes = document.querySelectorAll('[data-node-id]');
    console.log(`\n📦 找到 ${nodes.length} 个节点`);
    
    // 只监控前 5 个节点（避免日志过多）
    const nodesToMonitor = Array.from(nodes).slice(0, 5);
    nodesToMonitor.forEach((node, index) => {
        const nodeId = node.getAttribute('data-node-id');
        console.log(`📌 监控节点 ${index + 1}: ${nodeId}`);
        
        const initialState = {
            transform: node.style.transform,
            transition: node.style.transition
        };
        
        const observer = new MutationObserver((mutations) => {
            mutations.forEach(mutation => {
                if (mutation.attributeName === 'style') {
                    const timestamp = new Date().toISOString().split('T')[1];
                    const currentState = {
                        transform: node.style.transform,
                        transition: node.style.transition
                    };
                    
                    const changes = [];
                    if (currentState.transform !== initialState.transform) {
                        changes.push(`transform: "${initialState.transform}" → "${currentState.transform}"`);
                    }
                    if (currentState.transition !== initialState.transition) {
                        changes.push(`transition: "${initialState.transition}" → "${currentState.transition}"`);
                    }
                    
                    if (changes.length > 0) {
                        console.log(`\n⏰ [${timestamp}] 节点 ${nodeId} 样式变化:`);
                        changes.forEach(change => console.log(`  - ${change}`));
                        
                        // 🔥 检测闪烁的关键时刻
                        if (currentState.transition !== 'none' && currentState.transform === '') {
                            console.error(`🚨 节点闪烁警告！transition 已启用但 transform 被清除`);
                        }
                        
                        Object.assign(initialState, currentState);
                    }
                }
            });
        });
        
        observer.observe(node, { 
            attributes: true, 
            attributeFilter: ['style'] 
        });
    });
    
    console.log('\n✅ 监控器已完全启动');
    console.log('📝 现在拖动组，观察日志输出');
    console.log('🔥 特别注意 "闪烁警告" 的日志');
})();

🎯 关键检测点

  • 闪烁时刻: transition 已启用 + transform 被清除
  • 正常时刻: transition = 'none' + transform 被清除
  • 拖动中: transition = 'none' + transform 有值
  • 拖动结束: transition 恢复 + transform 清空（应该在同一帧）
