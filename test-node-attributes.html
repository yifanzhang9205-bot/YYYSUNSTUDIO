🔍 Node 组件属性验证工具

📋 使用说明

1. 确保你的应用正在运行（npm run dev）

2. 在应用页面打开开发者工具（F12）

3. 在控制台中复制粘贴下面的测试脚本

4. 按回车运行，查看结果

🧪 测试脚本 1: 基础检查

// ===== Node 属性验证脚本 1: 基础检查 =====
console.clear();
console.log('%c🔍 开始验证 Node 组件属性...', 'color: #00ff00; font-size: 16px; font-weight: bold;');
console.log('');

// 1. 查找所有带 data-node-id 的元素
const nodesWithDataAttr = document.querySelectorAll('[data-node-id]');
console.log(`%c✅ 找到 ${nodesWithDataAttr.length} 个带 data-node-id 的节点`, 'color: #00ff00; font-weight: bold;');

if (nodesWithDataAttr.length === 0) {
    console.log('%c❌ 错误：没有找到任何带 data-node-id 的节点！', 'color: #ff0000; font-weight: bold;');
    console.log('%c💡 可能的原因：', 'color: #ffaa00;');
    console.log('   1. 浏览器缓存未清除（按 Ctrl+Shift+R 硬刷新）');
    console.log('   2. Vite 开发服务器需要重启');
    console.log('   3. React 组件未正确渲染');
} else {
    console.log('');
    console.log('%c📊 节点详情：', 'color: #00aaff; font-weight: bold;');
    nodesWithDataAttr.forEach((node, index) => {
        const dataNodeId = node.getAttribute('data-node-id');
        const id = node.id;
        const hasDataAttr = node.hasAttribute('data-node-id');
        const hasIdAttr = node.hasAttribute('id');
        
        console.log(`%c节点 ${index + 1}:`, 'color: #ffaa00; font-weight: bold;');
        console.log(`  data-node-id: ${dataNodeId} ${hasDataAttr ? '✅' : '❌'}`);
        console.log(`  id: ${id} ${hasIdAttr ? '✅' : '❌'}`);
        console.log(`  className: ${node.className.substring(0, 50)}...`);
        console.log('');
    });
}

console.log('%c🎯 验证完成！', 'color: #00ff00; font-size: 16px; font-weight: bold;');

// ===== Node 属性验证脚本 2: DOM 查询测试 =====

    // 测试第一个节点
    const firstNode = allNodes[0];
    const testNodeId = firstNode.getAttribute('data-node-id');

    console.log(`  节点 ID: ${testNodeId}`);
    
    // 方法 1: 通过 data-node-id 查询
    const foundByDataAttr = document.querySelector(`[data-node-id="${testNodeId}"]`);
    console.log(`  方法 1 (data-node-id): ${foundByDataAttr ? '✅ 成功' : '❌ 失败'}`);
    
    // 方法 2: 通过 id 查询

    console.log(`  方法 2 (id): ${foundById ? '✅ 成功' : '❌ 失败'}`);
    
    // 方法 3: 通过 querySelector 查询 id
    const foundByQueryId = document.querySelector(`#node-${testNodeId}`);
    console.log(`  方法 3 (querySelector id): ${foundByQueryId ? '✅ 成功' : '❌ 失败'}`);
    
    console.log('');
    console.log('%c📊 查询结果对比：', 'color: #00aaff; font-weight: bold;');
    console.log('  原始节点:', firstNode);
    console.log('  data-node-id 查询:', foundByDataAttr);
    console.log('  id 查询:', foundById);
    console.log('  相同节点?', firstNode === foundByDataAttr && foundByDataAttr === foundById ? '✅ 是' : '❌ 否');
} else {
    console.log('%c❌ 没有节点可供测试！', 'color: #ff0000; font-weight: bold;');

console.log(`%c找到 ${allNodes.length} 个节点`, 'color: #00aaff;');

    console.log(`  失败: ${failCount} 个 ❌`);

console.log('%c🎯 测试完成！', 'color: #00ff00; font-size: 16px; font-weight: bold;');

🧪 测试脚本 4: 实时监控

// ===== Node 属性验证脚本 4: 实时监控 =====
console.clear();
console.log('%c🔍 启动实时监控...', 'color: #00ff00; font-size: 16px; font-weight: bold;');
console.log('%c按 Ctrl+C 停止监控', 'color: #ffaa00;');
console.log('');

let lastCount = 0;

const monitor = setInterval(() => {
    const nodes = document.querySelectorAll('[data-node-id]');
    const currentCount = nodes.length;
    
    if (currentCount !== lastCount) {
        console.log(`%c[${new Date().toLocaleTimeString()}] 节点数量变化: ${lastCount} → ${currentCount}`, 'color: #00aaff;');
        
        if (currentCount > 0) {
            console.log('%c  ✅ 所有节点都有 data-node-id 属性', 'color: #00ff00;');
        } else {
            console.log('%c  ❌ 没有找到带 data-node-id 的节点', 'color: #ff0000;');
        }
        
        lastCount = currentCount;
    }
}, 1000);

// 保存监控器引用，方便停止
window.nodeMonitor = monitor;
console.log('%c💡 提示：运行 clearInterval(window.nodeMonitor) 可以停止监控', 'color: #ffaa00;');

🔧 故障排除

如果测试失败，请按顺序尝试：

  1. 清除浏览器缓存
    • Windows/Linux: Ctrl + Shift + R
    • macOS: Cmd + Shift + R
  2. 重启 Vite 开发服务器
    • 在终端按 Ctrl + C 停止
    • 运行 npm run dev 重新启动
  3. 检查 React 渲染
    • 打开 React DevTools
    • 查看 Node 组件的 props
    • 确认 data-node-id 被传递
  4. 检查代码
    • 确认 Node.tsx 有 {...props}
    • 确认 App.tsx 传递了 data-node-id
    • 确认没有其他代码覆盖属性
