// 1. 在浏览器控制台中运行以下代码来注入监控： window.groupDragMonitor = { logs: [], timeline: [], stats: { dragCount: 0, flickerCount: 0, totalDuration: 0, rafCount: 0 }, log(type, message, data) { const entry = { timestamp: Date.now(), type, message, data }; this.logs.push(entry); // 发送到调试页面 if (window.opener) { window.opener.postMessage({ type: 'log', entry }, '*'); } }, startDrag(groupId) { this.stats.dragCount++; this.currentDrag = { groupId, startTime: Date.now(), events: [] }; this.log('info', '🚀 开始拖动', { groupId }); }, updateDrag(dx, dy) { if (this.currentDrag) { this.currentDrag.events.push({ time: Date.now() - this.currentDrag.startTime, dx, dy }); this.stats.rafCount++; } }, endDrag() { if (this.currentDrag) { const duration = Date.now() - this.currentDrag.startTime; this.stats.totalDuration += duration; this.log('info', '✅ 结束拖动', { duration: `${duration}ms`, events: this.currentDrag.events.length }); this.currentDrag = null; } }, detectFlicker(element) { // 监控元素的 transform 和 left/top 变化 const observer = new MutationObserver((mutations) => { mutations.forEach((mutation) => { if (mutation.type === 'attributes' && (mutation.attributeName === 'style')) { const style = element.style; const hasTransform = style.transform && style.transform !== 'none'; const hasPosition = style.left || style.top; if (hasTransform && hasPosition) { this.stats.flickerCount++; this.log('error', '⚠️ 检测到闪烁', { transform: style.transform, left: style.left, top: style.top, transition: style.transition }); } } }); }); observer.observe(element, { attributes: true, attributeFilter: ['style'] }); return observer; } }; // 2. 在 useGroup.ts 的关键位置添加监控调用： // startGroupDrag: window.groupDragMonitor?.startDrag(groupId); // updateGroupDrag: window.groupDragMonitor?.updateDrag(worldDx, worldDy); // endGroupDrag: window.groupDragMonitor?.endDrag(); // 3. 监控组元素： const groupElement = document.querySelector('[data-group-id="g-xxx"]'); if (groupElement) { window.groupDragMonitor?.detectFlicker(groupElement); }
